+ * primitives (issue 0807, and save.c RULING D5-1 states the rule).
+ *
+ * `xschem annotate_op` used to FREE the currently loaded 1-point op/dc database
+ * and only then try to open the new file. When the open failed -- ngspice mid
+ * rewrite, so the raw is readable but SHORT -- nothing replaced what had been
+ * deleted, the user's loaded database was simply gone, and the command still
+ * reported success. The fix is not "read into a scratch Raw and adopt it":
+ * read_rawfile_by_type() refuses any destination other than &xctx->raw for the
+ * non-spice types, and raw_read() on a foreign destination still runs
+ * set_modify(-2), the Waves cue repaint and the graph_flags & 4 backannotate
+ * block against the OLD xctx->raw. It is "take the entry OUT of the registry
+ * without freeing it, do the ordinary read, then free it (success) or put it
+ * back (failure)".

+ * Taking it out rather than leaving it in place is what keeps issue 0685 shut:
+ * extra_rawfile()'s read arm dedups on rawfile + sim_type and, on a hit,

+/* Remove the rawfile+sim_type entry from the registry WITHOUT freeing it, and
+ * return it; NULL if there is no such entry. `file` and `type` are compared
+ * verbatim (no `subst`, unlike extra_rawfile()'s lookups): the caller has the
+ * strings out of the entry itself. Post-state is the one the clear arm uses --
+ * indexes back to 0, xctx->raw at entry 0, or NULL when the registry empties.
+ * extra_raw_arr is NOT freed and extra_raw_size NOT zeroed even at n == 0,
+ * because something is going back in; every arm of extra_rawfile() guards on
+ * extra_raw_n, so "n == 0 with a live array" is a state they all tolerate. */
+Raw *extra_rawfile_detach(const char *file, const char *type)
+{
+  int i, found = -1;
+  Raw *raw;
+
+  if(!xctx || !file || !type || !xctx->extra_raw_arr) return NULL;
+  for(i = 0; i < xctx->extra_raw_n; i++) {
+    /* the same NULL skips as extra_rawfile()'s lookup loops (issue 0306) */
+    if(xctx->extra_raw_arr[i] && xctx->extra_raw_arr[i]->rawfile &&
+       xctx->extra_raw_arr[i]->sim_type &&
+       !strcmp(xctx->extra_raw_arr[i]->rawfile, file) &&
+       !strcmp(xctx->extra_raw_arr[i]->sim_type, type)) {
+      found = i;
+      break;
+    }
+  }
+  if(found < 0) return NULL;
+  raw = xctx->extra_raw_arr[found];
+  for(i = found; i < xctx->extra_raw_n - 1; i++) {
+    xctx->extra_raw_arr[i] = xctx->extra_raw_arr[i + 1];
+  }
+  xctx->extra_raw_n--;
+  xctx->extra_raw_arr[xctx->extra_raw_n] = NULL;
+  xctx->extra_idx = 0;
+  xctx->extra_prev_idx = 0;
+  xctx->raw = xctx->extra_raw_n ? xctx->extra_raw_arr[0] : NULL;
+  dbg(1, "extra_rawfile_detach(): %s %s, extra_raw_n=%d\n", file, type, xctx->extra_raw_n);
